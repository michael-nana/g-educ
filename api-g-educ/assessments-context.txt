    return this.cls.get('userId');
  }

  get role(): ClsStore['role'] {
    return this.cls.get('role');
  }

  get classId(): string | null | undefined {
    return this.cls.get('classId');
  }

  get status(): ClsStore['status'] {
    return this.cls.get('status');
  }

  get canPublish(): boolean | undefined {
    return this.cls.get('canPublish');
  }

  get ipAddress(): string | undefined {
    return this.cls.get('ipAddress');
  }

  setUserContext(context: Pick<
    ClsStore,
    'userId' | 'role' | 'classId' | 'status' | 'canPublish'

    this.cls.set('userId', context.userId);

import { NestFactory } from '@nestjs/core';
import { AppModule } from './app.module';
import { ValidationPipe } from '@nestjs/common';
import { DocumentBuilder, SwaggerModule } from '@nestjs/swagger';
import compression from 'compression';
import helmet from 'helmet';
import { HttpExceptionFilter } from './common/filters/http-exception.filter';
import { SerializationInterceptor } from './common/interceptors/serialization.interceptor';

async function bootstrap() {
  const app = await NestFactory.create(AppModule);
  app.setGlobalPrefix('api/v1');
  app.useGlobalPipes(
    new ValidationPipe({
      whitelist: true,
      forbidNonWhitelisted: true,
      transform: true,
    }),
  );
  app.useGlobalFilters(new HttpExceptionFilter());
  app.useGlobalInterceptors(new SerializationInterceptor());
  app.use(compression());
  app.use(helmet());
  app.enableCors({
    origin: process.env.CORS_ORIGINS?.split(',').map((origin) => origin.trim()) ?? false,
    credentials: true,
  });

  const config = new DocumentBuilder()
    .setTitle('G-Educ API')
    .setDescription('API backend de la plateforme G-Educ')
    .setVersion('1.0')
    .addBearerAuth()
    .build();
  const document = SwaggerModule.createDocument(app, config);
  SwaggerModule.setup('api/docs', app, document);

  await app.listen(process.env.PORT ?? 3000);
}
bootstrap();

import {
  BadRequestException,
  Injectable,
  NotFoundException,
} from '@nestjs/common';

import { PrismaService } from '../../../prisma/prisma.service';
import { AuditService } from '../../audit/services/audit.service';
import { AUDIT_ACTIONS } from '../../audit/constants/audit-actions.constants';

import { TeacherScopeService } from './teacher-scope.service';

import type { CreateChapterDto } from '../dto/create-chapter.dto';
import type { UpdateChapterDto } from '../dto/update-chapter.dto';
import type { ReorderChaptersDto } from '../dto/reorder-chapters.dto';

@Injectable()
export class ChaptersService {
  constructor(
    private readonly prisma: PrismaService,
    private readonly auditService: AuditService,
    private readonly teacherScope: TeacherScopeService,
  ) {}

  async create(
    user: { id: string; role: string },
    classSubjectId: string,
    dto: CreateChapterDto,
  ) {
    const classSubject = await this.teacherScope.assertCanManage(
      user,
      classSubjectId,
    );

    const title = dto.title.trim();

    if (!title) {
      throw new BadRequestException('Le titre est obligatoire');
    }

    return this.prisma.$transaction(async (tx) => {
      const chapter = await tx.chapter.create({
        data: {
          classSubjectId,
          classId: classSubject.classId,
          title,
          description: dto.description?.trim() || null,
          position: dto.position ?? 0,
          createdById: user.id,
        },
        select: {
          id: true,
          classSubjectId: true,
          classId: true,
          title: true,
          description: true,
          position: true,
          status: true,
          version: true,
          createdById: true,
          publishedAt: true,
          createdAt: true,
          updatedAt: true,
        },
      });

      await this.auditService.log(
        {
          action: AUDIT_ACTIONS.CHAPTER_CREATED,
          entityType: 'Chapter',
          entityId: chapter.id,
          after: {
            title: chapter.title,
            classId: chapter.classId,
            classSubjectId: chapter.classSubjectId,
          },
        },
        tx,
      );

      return chapter;
    });
  }

  async update(
    user: { id: string; role: string },
    chapterId: string,
    dto: UpdateChapterDto,
  ) {
    const chapter = await this.prisma.chapter.findUnique({
      where: { id: chapterId },
    });

    if (!chapter) {
      throw new NotFoundException('Chapitre introuvable');
    }

    await this.teacherScope.assertCanManage(
      user,
      chapter.classSubjectId,
    );

    if (
      dto.title === undefined &&
      dto.description === undefined &&
      dto.position === undefined
    ) {
      throw new BadRequestException(
        'Aucune modification fournie',
      );
    }

    const data = {
      ...(dto.title !== undefined && {
        title: dto.title.trim(),
      }),
      ...(dto.description !== undefined && {
        description: dto.description.trim() || null,
      }),
      ...(dto.position !== undefined && {
        position: dto.position,
      }),
      ...(chapter.status === 'PUBLIE' && {
        version: { increment: 1 },
      }),
    };

    if (dto.title !== undefined && !dto.title.trim()) {
      throw new BadRequestException('Le titre est obligatoire');
    }

    return this.prisma.$transaction(async (tx) => {
      const updated = await tx.chapter.update({
        where: { id: chapterId },
        data,
      });

      await this.auditService.log(
        {
          action: AUDIT_ACTIONS.CHAPTER_UPDATED,
          entityType: 'Chapter',
          entityId: chapterId,
          before: {
            title: chapter.title,
            description: chapter.description,
            position: chapter.position,
            version: chapter.version,
          },
          after: {
            title: updated.title,
            description: updated.description,
            position: updated.position,
            version: updated.version,
          },
        },
        tx,
      );

      return updated;
    });
  }

  async publish(
    user: { id: string; role: string; canPublish: boolean },
    chapterId: string,
  ) {
    const chapter = await this.prisma.chapter.findUnique({
      where: { id: chapterId },
    });

    if (!chapter) {
      throw new NotFoundException('Chapitre introuvable');
    }

    await this.teacherScope.assertCanManage(
      user,
      chapter.classSubjectId,
    );

    if (
      user.role !== 'ADMINISTRATEUR' &&
      !user.canPublish
    ) {
      throw new BadRequestException(
        'Vous ne disposez pas du droit de publication',
      );
    }

    if (chapter.status === 'PUBLIE') {
      throw new BadRequestException(
        'Ce chapitre est dÃ©jÃ  publiÃ©',
      );
    }

    return this.prisma.$transaction(async (tx) => {
      const updated = await tx.chapter.update({
        where: { id: chapterId },
        data: {
          status: 'PUBLIE',
          publishedAt: new Date(),
        },
      });

      await this.auditService.log(
        {
          action: AUDIT_ACTIONS.CHAPTER_PUBLISHED,
          entityType: 'Chapter',
          entityId: chapterId,
          before: {
            status: chapter.status,
          },
          after: {
            status: updated.status,
            publishedAt: updated.publishedAt?.toISOString(),
          },
        },
        tx,
      );

      return updated;
    });
  }

  async archive(
    user: { id: string; role: string },
    chapterId: string,
  ) {
    const chapter = await this.prisma.chapter.findUnique({
      where: { id: chapterId },
    });

    if (!chapter) {
      throw new NotFoundException('Chapitre introuvable');
    }

    await this.teacherScope.assertCanManage(
      user,
      chapter.classSubjectId,
    );

    if (chapter.status === 'ARCHIVE') {
      throw new BadRequestException(
        'Ce chapitre est dÃ©jÃ  archivÃ©',
      );
    }

    return this.prisma.$transaction(async (tx) => {
      const updated = await tx.chapter.update({
        where: { id: chapterId },
        data: {
          status: 'ARCHIVE',
        },
      });

      await this.auditService.log(
        {
          action: AUDIT_ACTIONS.CHAPTER_ARCHIVED,
          entityType: 'Chapter',
          entityId: chapterId,
          before: {
            status: chapter.status,
          },
          after: {
            status: updated.status,
          },
        },
        tx,
      );

      return updated;
    });
  }

  async reorder(
    user: { id: string; role: string },
    classSubjectId: string,
    dto: ReorderChaptersDto,
  ) {
    await this.teacherScope.assertCanManage(
      user,
      classSubjectId,
    );

    const chapters = await this.prisma.chapter.findMany({
      where: { classSubjectId },
      select: {
        id: true,
        position: true,
      },
    });

    const requestedIds = dto.chapterIds;

    if (
      chapters.length !== requestedIds.length ||
      chapters.some(
        (chapter) => !requestedIds.includes(chapter.id),
      )
    ) {
      throw new BadRequestException(
        'La liste doit contenir exactement tous les chapitres de cette matiÃ¨re',
      );
    }

    await this.prisma.$transaction(async (tx) => {
      for (const [index, id] of requestedIds.entries()) {

export interface CreateAuditLogInput {
  action: string;
  entityType: string;
  entityId?: string | null;
  before?: Prisma.InputJsonValue | Prisma.NullableJsonNullValueInput;
  after?: Prisma.InputJsonValue | Prisma.NullableJsonNullValueInput;
}

@Injectable()
export class AuditService {
  constructor(
    private readonly prisma: PrismaService,
    private readonly clsContext: ClsContextService,
  ) {}

  async findAll(filters: AuditQueryDto) {
  const {
    userId,
    entityType,
    entityId,
    action,
    from,
    to,
    page = 1,
    limit = 20,
  } = filters;

  const where: Prisma.AuditLogWhereInput = {
    ...(userId && { userId }),
    ...(entityType && { entityType }),
    ...(entityId && { entityId }),
    ...(action && { action }),
    ...(from || to
      ? {
          createdAt: {
            ...(from && { gte: new Date(from) }),
            ...(to && { lte: new Date(to) }),
          },
        }
      : {}),
  };

  const [data, total] = await this.prisma.$transaction([
    this.prisma.auditLog.findMany({
      where,
      orderBy: {
        createdAt: 'desc',
      },
      skip: (page - 1) * limit,
      take: limit,
    }),
    this.prisma.auditLog.count({ where }),
  ]);

const serializedData = data.map((log) => ({
  ...log,
  id: log.id.toString(),
}));

return {
  data: serializedData,
  meta: {
    total,
    page,
    limit,
    totalPages: Math.ceil(total / limit),
  },
};
}

  async log(
    input: CreateAuditLogInput,
    tx?: Prisma.TransactionClient,
  ) {
    const client = tx ?? this.prisma;

    return client.auditLog.create({
      data: {
        userId: this.clsContext.userId ?? null,
        role: this.clsContext.role ?? null,
        action: input.action,
        entityType: input.entityType,
        entityId: input.entityId ?? null,
        before: input.before,
        after: input.after,
        ipAddress: this.clsContext.ipAddress ?? null,
      },
    });
  }

  ExecutionContext,
  Injectable,
} from '@nestjs/common';
import { Reflector } from '@nestjs/core';
import type { Role } from '../../../generated/prisma/client';
import type { AuthenticatedUser } from '../decorators/current-user.decorator';
import { ROLES_KEY } from '../decorators/roles.decorator';

@Injectable()
export class RolesGuard implements CanActivate {
  constructor(private readonly reflector: Reflector) {}

  canActivate(context: ExecutionContext): boolean {
    const requiredRoles = this.reflector.getAllAndOverride<Role[]>(
      ROLES_KEY,
      [context.getHandler(), context.getClass()],
    );

    if (!requiredRoles || requiredRoles.length === 0) {
      return true;
    }

    const request = context
      .switchToHttp()
      .getRequest<{ user?: AuthenticatedUser }>();

    const user = request.user;

    if (!user) {
      return false;
    }

    return requiredRoles.includes(user.role as Role);
  }
}

  async uploadPdfAsset(
  user: CurrentUser,
  lessonId: string,
  file: Express.Multer.File,
) {
  if (!file) {
    throw new BadRequestException('Aucun fichier fourni.');
  }

  if (!file.buffer || file.buffer.length === 0) {
    throw new BadRequestException('Le fichier est vide ou invalide.');
  }

  if (file.size > MAX_PDF_SIZE) {
    throw new BadRequestException(
      'Le fichier PDF ne doit pas dÃ©passer 20 Mo.',
    );
  }

  // VÃ©rification du contenu rÃ©el du fichier, et non de son extension.
  const fileType = await fileTypeFromBuffer(file.buffer);

  if (fileType?.mime !== 'application/pdf') {
    throw new BadRequestException(
      'Le fichier fourni nâ€™est pas un vÃ©ritable PDF.',
    );
  }

  const lesson = await this.findLessonOrFail(lessonId);

  await this.teacherScopeService.assertCanManage(
    user,
    lesson.chapter.classSubjectId,
  );

  if (lesson.type !== LessonType.PDF) {
    throw new BadRequestException(
      'Les fichiers PDF ne peuvent Ãªtre ajoutÃ©s quâ€™aux leÃ§ons de type PDF.',
    );
  }

  if (lesson.status === ContentStatus.ARCHIVE) {
    throw new BadRequestException(
      'Impossible de modifier une leÃ§on archivÃ©e.',
    );
  }

  const storageKey =
    `pdf/${lesson.classId}/${randomUUID()}.pdf`;

  // 1. Upload dans Backblaze B2.
  await this.uploadService.uploadFile({
    storageKey,
    buffer: file.buffer,
    mimeType: 'application/pdf',
  });

  try {
    // 2. Enregistrement en base et journalisation.
    return await this.prisma.$transaction(async (tx) => {
      const asset = await tx.lessonAsset.create({
        data: {
          lessonId: lesson.id,
          kind: AssetKind.PDF,
          storageKey,
          mimeType: 'application/pdf',
          sizeBytes: file.buffer.length,
          processingStatus: AssetProcessingStatus.PRET,
        },
      });

      if (lesson.status === ContentStatus.PUBLIE) {
        await tx.lesson.update({
          where: { id: lesson.id },
          data: { version: { increment: 1 } },
        });
      }

      await this.auditService.log(
        {
          action: AUDIT_ACTIONS.ASSET_UPLOADED,
          entityType: 'LESSON_ASSET',
          entityId: asset.id,
          after: {
            lessonId: lesson.id,
            kind: AssetKind.PDF,
            sizeBytes: asset.sizeBytes,
          },
        },
        tx,
      );

      return asset;
    });
  } catch (error) {
    // 3. Si l'enregistrement Ã©choue, on supprime l'objet B2.
    try {
      await this.uploadService.deleteFile(storageKey);
    } catch {
      // Ne pas masquer l'erreur initiale de la transaction.
    }

    throw error;
  }
}

import { Injectable, OnModuleDestroy, OnModuleInit } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import { PrismaPg } from '@prisma/adapter-pg';
import { PrismaClient } from '../generated/prisma/client';
import { ClsService } from 'nestjs-cls';
import { ForbiddenException } from '@nestjs/common';
import type { ClsStore } from '../common/cls/cls-store.interface';

@Injectable()
export class PrismaService
  extends PrismaClient
  implements OnModuleInit, OnModuleDestroy
{
  readonly scopedClient;

  constructor(config: ConfigService, cls: ClsService<ClsStore>) {
    const adapter = new PrismaPg({
      connectionString: config.getOrThrow<string>('DATABASE_URL'),
    });
    super({ adapter });

    const classScopedModels = [
      'ClassSubject',
      'Chapter',
      'Lesson',
      'Assessment',
      'Event',
      'Contribution',
      'ForumThread',
      'ForumAnswer',
    ];

    this.scopedClient = this.$extends({
      query: {
        $allModels: {
          async $allOperations({ model, operation, args, query }) {
            if (!classScopedModels.includes(model)) {
              return query(args);
            }

            if (cls.get('role') !== 'ELEVE') {
              return query(args);
            }

            const classId = cls.get('classId');
            if (!classId) {
              throw new ForbiddenException('Student class is missing from context');
            }

            const operationArgs = args as {
              data?: Record<string, unknown> | Array<Record<string, unknown>>;
              where?: Record<string, unknown>;
            };

            if (operation === 'create' || operation === 'createMany') {
              if (operation === 'create') {
                operationArgs.data = {
                  ...(operationArgs.data as Record<string, unknown> | undefined),
                  classId,
                };
              } else if (Array.isArray(operationArgs.data)) {
                operationArgs.data = operationArgs.data.map((item) => ({ ...item, classId }));
              } else if (operationArgs.data) {
                operationArgs.data = {
                  ...(operationArgs.data as Record<string, unknown>),
                  classId,
                };
              }
              return query(args);
            }

            if (operationArgs.where) {
              operationArgs.where = { ...operationArgs.where, classId };
            } else {
              operationArgs.where = { classId };
            }
            return query(args);
          },
        },
      },
    });
  }

  async onModuleInit() {
    await this.$connect();
  }

  async onModuleDestroy() {
    await this.$disconnect();
  }
}
